Python SDK Example
import omnisync

client = omnisync.Client(
    api_key="os_live_KEY"
)

signals = client.finance.signals.list(
    signal_type="clinical_catalyst",
    start_date="2024-01-01",
    ticker="MRNA"
)

for s in signals.data:
    print(s.direction, s.confidence)
